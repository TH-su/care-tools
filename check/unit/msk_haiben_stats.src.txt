function fmtDur(h){
  if(h===null||h===undefined||isNaN(h)||h<0)return'—';

  var d=Math.floor(tm/1440),hr=Math.floor((tm%1440)/60),mn=tm%60;

  if(hr>0)return hr+'時間'+(mn>0?mn+'分':'');
  return mn+'分';
}

      (re.laxNote?'<div>下剤メモ: '+escapeHtmlText(String(re.laxNote).replace(/\n/g,' '))+'</div>':'');
    return '<div class="rp-pg">'+
      rpHead('排便の経過（主治医・訪問看護への情報提供）',
             re.room+'号室　'+re.name,
             '対象期間: '+from+' 〜 '+to+'（'+dates.length+'日）')+
      '<div class="rp-sum">'+sum+'</div>'+
      '<table class="rp-t rp-t-bo"><thead><tr>'+
        '<th>日</th><th>曜</th><th>排便（時刻・性状・量）</th><th>下剤と反応</th><th>水分<br>(cc)</th><th>排尿<br>(回)</th><th>備考（排便・下剤）</th>'+
      '</tr></thead><tbody>'+rows+'</tbody></table>'+
      rpFoot('')+'</div>';
  }).join('');
